1|القول في تأويل قوله تعالى: ﴿وَالسَّمَاءِ وَالطَّارِقِ (١) وَمَا أَدْرَاكَ مَا الطَّارِقُ (٢) النَّجْمُ الثَّاقِبُ (٣) إِنْ كُلُّ نَفْسٍ لَمَّا عَلَيْهَا حَافِظٌ (٤) فَلْيَنْظُرِ الإنْسَانُ مِمَّ خُلِقَ (٥) خُلِقَ مِنْ مَاءٍ دَافِقٍ (٦) يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ (٧) إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ (٨) يَوْمَ تُبْلَى السَّرَائِرُ (٩) فَمَا لَهُ مِنْ قُوَّةٍ وَلا نَاصِرٍ (١٠) ﴾\n.\nأقسم ربنا بالسماء وبالطارق الذي يطرق ليلا من النجوم المضيئة، ويخفى نهارًا، وكل ما جاء ليلا فقد طرق.\nوبنحو الذي قلنا في ذلك قال أهل التأويل.\nذكر من قال ذلك:\n⁕ حدثني محمد بن سعد، قال: ثني أبي، قال: ثني عمي، قال: ثني أبي، عن أبيه، عن ابن عباس ﴿وَالسَّمَاءِ وَالطَّارِقِ﴾ قال: السماء وما يطرق فيها.\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة ﴿وَالسَّمَاءِ وَالطَّارِقِ وَمَا أَدْرَاكَ مَا الطَّارِقُ﴾ قال: طارق يطرق بليل، ويخفى بالنهار.\n⁕ حدثنا ابن عبد الأعلى، قال: ثنا ابن ثور، عن معمر، عن قتادة، في قوله: ﴿وَالطَّارِقِ﴾ قال: ظهور النجوم، يقول: يطرقك ليلا.\n⁕ حُدثت عن الحسين، قال: سمعت أبا معاذ يقول: ثنا عبيد، قال: سمعت الضحاك يقول في قوله: ﴿الطَّارِقِ﴾ النجم.\n﴿وَمَا أَدْرَاكَ مَا الطَّارِقُ﴾ يقول تعالى ذكره لنبيه محمد ﷺ: وما أشعرك يا محمد ما الطارق الذي أقسمتُ به، ثم بين ذلك جلّ ثناؤه، فقال: هو النجم الثاقب، يعني: يتوقد ضياؤه ويتوهَّج.\nوبنحو الذي قلنا في ذلك قال أهل التأويل.\nذكر من قال ذلك:\n⁕ حدثني عليّ، قال: ثنا أبو صالح، قال: ثني معاوية، عن عليّ، عن ابن عباس، في قوله: ﴿النَّجْمُ الثَّاقِبُ﴾ يعني: المضيء.\n⁕ حدثني محمد بن سعد، قال: ثني أبي، قال: ثني عمي، قال: ثني أبي، عن أبيه، عن ابن عباس ﴿النَّجْمُ الثَّاقِبُ﴾ قال: هي الكواكب المضيئة، وثقوبه: إذا أضاء.\n⁕ حدثنا ابن حميد، قال: ثنا يحيى بن واضح، قال: ثنا الحسين، عن يزيد، عن عكرِمة، في قوله: ﴿النَّجْمُ الثَّاقِبُ﴾ قال: الذي يثقب.\n⁕ حدثني محمد بن عمرو، قال: ثنا أبو عاصم، قال: ثنا عيسى، وحدثني الحارث، قال: ثنا الحسن قال: ثنا ورقاء، جميعا عن ابن أبي نجيح، عن مجاهد، في قول الله: ﴿الثَّاقِبُ﴾ قال: الذي يتوهَّج.\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة: ثقوبه: ضوءه.\n⁕ حدثنا ابن عبد الأعلى، قال: ثنا ابن ثور، عن معمر، عن قتادة ﴿النَّجْمُ الثَّاقِبُ﴾: المضيء.\n⁕ حدثني يونس، قال: أخبرنا ابن وهب، قال: قال ابن زيد، في قوله: ﴿النَّجْمُ الثَّاقِبُ﴾ قال: كانت العرب تسمِّي الثُّريا النجم، ويقال: إن الثاقبَ النجمُ الذي يقال له زُحَلْ. والثاقب أيضًا: الذي قد ارتفع على النجوم، والعرب تقول للطائر - إذا هو لحق ببطن السماء ارتفاعًا -: قد ثَقَبَ، والعرب تقول: أثقِب نارك: أي أضئها.\n* * وقوله: ﴿إِنْ كُلُّ نَفْسٍ لَمَّا عَلَيْهَا حَافِظٌ﴾\nاختلفت القرّاء في قراءة ذلك، فقرأه من قرّاء المدينة أبو جعفر، ومن قرّاء الكوفة حمزة ﴿لَمَّا عَلَيْهَا﴾ بتشديد الميم. وذُكِر عن الحسن أنه قرأ ذلك كذلك.\n⁕ حدثني أحمد بن يوسف، قال: ثنا أبو عبيد، قال: ثنا حجاج، عن هارون، عن الحسن أنه كان يقرؤها ﴿إِنْ كُلُّ نَفْسٍ لَمَّا عَلَيْهَا حَافِظٌ﴾ مشدّدة، ويقول: إلا ليها حافظ، وهكذا كلّ شيءٍ في القرآن بالتثقيل. وقرأ ذلك من أهل المدينة نافع، ومن أهل البصرة أبو عمرو: ﴿لَمَا﴾ بالتخفيف، بمعنى: إن كلّ نفس لعليها حافظ، وعلى أن اللام جواب" إن" و" ما" التي بعدها صلة. وإذا كان ذلك كذلك لم يكن فيه تشديد.\nوالقراءة التي لا أختار غيرَها في ذلك التخفيف، لأن ذلك هو الكلام المعروف من كلام العرب، وقد أنكر التشديد جماعةٌ من أهل المعرفة بكلام العرب؛ أن يكون معروفًا من كلام العرب، غير أن الفرّاء كان يقول: لا نعرف جهة التثقيل في ذلك، ونرى أنها لغةٌ في هذيلٍ، يجعلون " إلا " مع " إن المخففة ": لَمَّا، ولا يجاوزون ذلك، كأنه قال: ما كلّ نفس إلا عليها حافظ، فإن كان صحيحًا ما ذكر الفرّاء من أنها لغة هُذَيلٍ، فالقراءة بها جائزةٌ صحيحةٌ، وإن كان الاختيار أيضًا - إذا صحّ ذلك عندنا - القراءة الأخرى وهي التخفيف؛ لأن ذلك هو المعروف من كلام العرب، ولا ينبغي أن يُترك الأعرف إلى الأنكر.\n⁕ وقد حدثني أحمد بن يوسف، قال: ثنا أبو عبيد، قال: ثنا معاذ، عن ابن عون، قال: قرأت عند ابن سيرين: ﴿إِنْ كُلُّ نَفْسٍ لَمَّا عَلَيْهَا حَافِظٌ﴾ فأنكره، وقال: سبحان الله، سبحان الله.\nفتأويل الكلام إذن: إن كلّ نفس لعليها حافظ من ربها، يحفظ عملها، ويُحصي عليها ما تكسب من خير أو شرّ.\nوبنحو الذي قلنا في ذلك قال أهل التأويل.\nذكر من قال ذلك:\n⁕ حدثني محمد بن سعد، قال: ثني أبي، قال: ثني عمي، قال: ثني أبي، عن أبيه، عن ابن عباس، قوله: ﴿إنْ كُلُّ نَفْسٍ لَمَا عَلَيْهَا حَافِظٌ﴾ قال: كلّ نفس عليها حفظة من الملائكة.\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة، قوله: ﴿إنْ كُلُّ نَفْسٍ لَمَا عَلَيْهَا حَافِظٌ﴾: حفظة يحفظون عملك ورزقك وأجلك، إذا توفيته يا ابن آدم قُبِضت إلى ربك.\n* * وقوله: ﴿فَلْيَنْظُرِ الإنْسَانُ مِمَّ خُلِقَ﴾\nيقول تعالى ذكره: فلينظر الإنسان المكذّب بالبعث بعد الممات، المُنكر قُدرة الله على إحيائه بعد مماته، ﴿مِمَّ خُلِقَ﴾ يقول: من أيّ شيءٍ خلقه ربه، ثم أخبر جلّ ثناؤه عما خلقه منه، فقال: (خُلِقَ مِنْ مَاءٍ دَافِقٍ) يعني: من ماء مدفوق، وهو مما أخرجته العرب بلفظ فاعل، وهو بمعنى المفعول، ويقال: إن أكثر من يستعمل ذلك من أحياء العرب سكان الحجاز إذا كان في مذهب النعت، كقولهم: هذا سرٌّ كاتم وهمٌّ ناصب، ونحو ذلك.\n* * وقوله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾\nيقول: يخرج من بين ذلك، ومعنى الكلام: منهما، كما يقال: سيخرج من بين هذين الشيئين خير كثير، بمعنى: يخرج منهما.\nواختلف أهل التأويل في معنى الترائب وموضعها، فقال بعضهم: الترائب: موضع القلادة من صدر المرأة.\nذكر من قال ذلك:\n⁕ حدثني عبد الرحمن بن الأسود الطفاويّ، قال: ثنا محمد بن ربيعة، عن سَلَمَة بن سابور، عن عطية العَوْفِيّ، عن ابن عباس ﴿الصُّلْبِ وَالتَّرَائِبِ﴾ قال: الترائب: موضع القِلادة.\n⁕ حدثني عليّ، قال: ثنا أبو صالح، قال: ثني معاوية، عن عليّ، عن ابن عباس، قوله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ يقول: من بين ثدي المرأة.\n⁕ حدثني يعقوب، قال: ثنا ابن علية، عن أبي رجاء، قال: سُئل عكرِمة عن الترائب، فقال: هذه، ووضع يده على صدره بين ثدييه.\n⁕ حدثني ابن المثنى، قال: ثني سلم بن قتيبة، قال: ثني عبد الله بن النُّعمان الحُدَانيّ، أنه سمع عكرِمة يقول: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ قال: صُلْب الرجل، وترائب المرأة.\n⁕ حدثنا أبو كريب، قال: ثنا ابن يمان، عن شريك، عن عطاء، عن سعيد بن جبير، قال: الترائب الصدر.\n⁕ قال: ثنا ابن يمان، عن مِسْعَر، عن الحكم، عن أبي عياض، قال: ﴿التَّرَائِبِ﴾: الصدر.\n⁕ حدثني يونس، قال: أخبرنا ابن وهب، قال: قال ابن زيد، في قوله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ قال: الترائب: الصدر. وهذا الصلب وأشار إلى ظهره.\nوقال آخرون: الترائب: ما بين المنكبين والصدر.\nذكر من قال ذلك:\n⁕ حدثنا أبو كريب، قال: ثنا ابن يمان، عن إسرائيل، عن ثُوَير، عن مجاهد، قال: ﴿الترائِبِ﴾ ما بين المنكبين والصدر.\n⁕ حدثني محمد بن عمرو، قال: ثنا أبو عاصم، قال: ثنا عيسى، وحدثني الحارث، قال: ثنا الحسن، قال: ثنا ورقاء، جميعا عن ابن أبي نجيح، عن مجاهد، قوله: ﴿التَّرَائِبِ﴾ قال: أسفل من التراقي.\n⁕ حدثنا ابن حميد، قال: ثنا مهران، عن سفيان، قال: الصُّلْب للرجل، والترائب للمرأة، والترائب فوق الثديين.\nوقال آخرون: هو اليدان والرجلان والعينان.\nذكر من قال ذلك:\n⁕ حدثني محمد بن سعد، قال: ثني أبي، قال: ثني عمي، قال: ثني أبي، عن أبيه، عن ابن عباس، قوله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ قال: فالترائب أطراف الرجل واليدان والرجلان والعينان، فتلك الترائب.\n⁕ حدثنا ابن حميد، قال: ثنا مهران، عن سفيان، عن أبي رَوْق، عن الضحاك ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ قال: الترائب: اليدان والرجلان.\n⁕ قال: ثنا مهران، عن سفيان، قال: قال غيره: الترائب: ماء المرأة وصلب الرجل.\n⁕ حُدثت عن الحسين، قال: سمعت أبا معاذ يقول: ثنا عبيد، قال: سمعت الضحاك يقول في قوله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾: عيناه ويداه ورجلاه.\nوقال آخرون: معنى ذلك، أنه يخرج من بين صلب الرجل ونحره.\nذكر من قال ذلك:\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة، قوله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ يقول: يخرج من بين صلب الرجل ونحره.\nوقال آخرون: هي الأضلاع التي أسفل الصلب.\nذكر من قال ذلك:\n⁕ حدثنا أبو كُرَيب، قال: ثنا ابن يمان، عن أشعث، عن جعفر، عن سعيد، في قوله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ قال: الترائب: الأضلاع التي أسفل الصلب.\nوقال آخرون: هي عصارة القلب.\nذكر من قال ذلك:\n⁕ حدثني محمد بن إسحاق، قال: ثنا أبو صالح، قال: ثني الليث، أن معمر بن أبي حَبِيبة المَدِيني حدّثه، أنه بلغه في قول الله: ﴿يَخْرُجُ مِنْ بَيْنِ الصُّلْبِ وَالتَّرَائِبِ﴾ قال: هو عُصَارة القلب، ومنه يكون الولد.\nوالصواب من القول في ذلك عندنا، قولُ من قال: هو موضع القِلادة من المرأة، حيث تقع عليه من صدرها؛ لأن ذلك هو المعروف في كلام العرب، وبه جاءت أشعارهم، قال المثقَّب العبدي:\nوَمِنْ ذَهَبٍ يُسَنُّ عَلَى تَرِيبٍ . كَلَوْنِ العَاجِ لَيْسَ بِذِي غُضُونِ وقال آخر:\nوَالزَّعْفَرَانُ عَلَى تَرَائِبِهَا شَرِقا . بِهِ اللَّبَّاُت والنَّحْرُ * * وقوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾\nيقول تعالى ذكره: إن هذا الذي خلقكم أيها الناس من هذا الماء الدافق، فجعلكم بشرًا سويًا، بعد أن كنتم ماء مدفوقًا، على رجعه لقادر.\nواختلف أهل التأويل في الهاء التي في قوله: ﴿عَلَى رَجْعِهِ﴾ على ما هي عائدةٌ، فقال بعضهم: هي عائدة على الماء. وقالوا: معنى الكلام: إن الله على ردّ النطفة في الموضع التي خرجت منه ﴿لَقَادِرٌ﴾.\nذكر من قال ذلك:\n⁕ حدثني يعقوب، قال: ثنا ابن عُلَية، عن أبي رجاء، عن عكرِمة، في قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ قال: إنه على رَدّه في صُلْبه لقادر.\n⁕ حدثنا ابن المثنى، قال: ثنا أبو النعمان الحكم بن عبد الله، قال: ثنا شعبة، عن أبي رجاء، عن عكرمة في قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ قال: لِلصُّلب.\n⁕ حدثني عبيد بن إسماعيل الهباري، قال: ثنا عبد الرحمن بن محمد المحاربيّ، عن ليث، عن مجاهد، في قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ قال: على أن يرد الماء في الإحليل.\n⁕ حدثني نصر بن عبد الرحمن الأوْدِيّ الوشاء، قال: ثنا أبو قَطَن عمرو بن الهيثم، عن ورقاء، عن عبد الله بن أبي نجيح، عن عبد الله بن أبي بكر، عن مجاهد، في قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ قال: على ردّ النطفة في الإحليل.\n⁕ حدثني محمد بن عمرو، قال: ثنا أبو عاصم، قال: ثنا عيسى، وحدثني الحارث، قال: ثنا الحسن، قال: ثنا ورقاء، جميعا عن ابن أبي نجيح، عن مجاهد، قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ قال: في الإحليل.\n⁕ حدثنا ابن حميد، قال: ثنا مهران، عن سفيان، عن ليث، عن مجاهد ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ قال: ردّه في الإحليل.\nوقال آخرون: بل معنى ذلك: إنه على ردّ الإنسان ماءً كما كان قبل أن يخلقه منه.\nذكر من قال ذلك:\n⁕ حُدثت عن الحسين، قال: سمعت أبا معاذ يقول: ثنا عبيد، قال: سمعت الضحاك يقول في قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ إن شئتُ رددتُه كما خلقته من ماءٍ.\nوقال آخرون: بل معنى ذلك: إنه على حبس ذلك الماء لقادرٌ.\nذكر من قال ذلك:\n⁕ حدثني يونس، قال: أخبرنا ابن وهب، قال: قال ابن زيد، في قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ قال: على رجع ذلك الماء لقادرٌ، حتى لا يخرج، كما قدر على أن يخلق منه ما خلق قادر على أن يرجعه.\nوقال آخرون: بل معنى ذلك أنه قادر على رجع الإنسان من حال الكبر إلى حال الصغر.\nذكر من قال ذلك:\n⁕ حدثنا ابن حميد، قال: ثنا يحيى بن واضح، قال: ثنا الحسين، عن مقاتل بن حيان، عن الضحاك قال: سمعته يقول في قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ يقول: إن شئتُ رددته من الكبر إلى الشباب، ومن الشباب إلى الصِّبا، ومن الصبا إلى النطفة. وعلى هذا التأويل تكون الهاء في قوله: ﴿عَلَى رَجْعِهِ﴾ من ذكر الإنسان.\nوقال آخرون، ممن زعم أن الهاء للإنسان: معنى ذلك أنه على إحيائه بعد مماته لقادر.\nذكر من قال ذلك:\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة، قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ إن الله تعالى ذكره على بعثه وإعادته قادر.\nوأولى الأقوال في ذلك بالصواب قول من قال: معنى ذلك: إن الله على ردّ الإنسان المخلوق من ماء دافق من بعد مماته حيًّا، كهيئته قبل مماته لقادر.\nوإنما قلت: هذا أولى الأقوال في ذلك بالصواب؛ لقوله: ﴿يَوْمَ تُبْلَى السَّرَائِرُ﴾ فكان في إتباعه قوله: ﴿إِنَّهُ عَلَى رَجْعِهِ لَقَادِرٌ﴾ نبأ من أنباء القيامة، دلالة على أن السابق قبلها أيضًا منه، ومنه ﴿يَوْمَ تُبْلَى السَّرَائِرُ﴾ يقول تعالى ذكره: إنه على إحيائه بعد مماته لقادر يوم تُبلى السرائر، فاليوم من صفة الرجع، لأن المعنى: إنه على رجعه يوم تبلى السرائر لقادر.\nوَعُنِي بقوله: ﴿يَوْمَ تُبْلَى السَّرَائِرُ﴾\nيوم تُخْتَبَرُ سرائر العباد، فيظهر منها يومئذ ما كان في الدنيا مستخفيًا عن أعين العباد من الفرائض التي كان الله ألزمه إياها، وكلَّفه العمل بها.\nوبنحو الذي قلنا في ذلك قال أهل التأويل.\nذكر من قال ذلك:\n⁕ حُدثت عن عبد الله بن صالح، عن يحيى بن أيوب، عن ابن جريج، عن عطاء بن أبي رباح، في قوله: ﴿يَوْمَ تُبْلَى السَّرَائِرُ﴾ قال: ذلك الصوم والصلاة وغسل الجنابة، وهو السرائر، ولو شاء أن يقول: قد صُمْتُ، وليس بصائم، وقد صلَّيتُ، ولم يصلّ، وقد اغتسلت، ولم يغتسل.\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة ﴿يَوْمَ تُبْلَى السَّرَائِرُ﴾ إن هذه السرائر مختبرة، فأسِرّوا خيرًا وأعلنوه إن استطعتم، ولا قوّة إلا بالله.\n⁕ حدثنا ابن حميد، قال: ثنا مهران، عن سفيان ﴿يَوْمَ تُبْلَى السَّرَائِرُ﴾ قال: تُخْتَبَر.\n* * وقوله: ﴿فَمَا لَهُ مِنْ قُوَّةٍ وَلا نَاصِرٍ﴾\nيقول تعالى ذكره: فما للإنسان الكافر يومئذ من قوّة يمتنع بها من عذاب الله، وأليم نكاله، ولا ناصر ينصره فيستنقذه ممن ناله بمكروه، وقد كان في الدنيا يرجع إلى قوّةٍ من عشيرته، يمتنع بهم ممن أراده بسوءٍ، وناصرٍ من حليفٍ ينصره على من ظلمه واضطهده.\nوبنحو الذي قلنا في ذلك قال أهل التأويل.\nذكر من قال ذلك:\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة، قوله: ﴿فَمَا لَهُ مِنْ قُوَّةٍ وَلا نَاصِرٍ﴾ ينصره من الله.\n⁕ حدثنا ابن عبد الأعلى، قال: ثنا ابن ثور، عن معمر، عن قتادة، في قوله: ﴿وَلا نَاصِرٍ﴾ قال: من قوّة يمتنع بها، ولا ناصر ينصره من الله.\n⁕ حدثني عليّ بن سهل، قال: ثنا ضَمْرة بن ربيعة، عن سفيان الثوري، في قوله: ﴿مِنْ قُوَّةٍ وَلا نَاصِرٍ﴾ قال: القوّة: العشيرة، والناصر: الحليف.

11|القول في تأويل قوله تعالى: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ (١١) وَالأرْضِ ذَاتِ الصَّدْعِ (١٢) إِنَّهُ لَقَوْلٌ فَصْلٌ (١٣) وَمَا هُوَ بِالْهَزْلِ (١٤) إِنَّهُمْ يَكِيدُونَ كَيْدًا (١٥) وَأَكِيدُ كَيْدًا (١٦) فَمَهِّلِ الْكَافِرِينَ أَمْهِلْهُمْ رُوَيْدًا (١٧) ﴾\n.\nيقول تعالى ذكره: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ ترْجع بالغيوم وأرزاق العباد كلّ عام؛ ومنه قول المتنخِّل في صفة سيف:\nأبْيَضُ كالرَّجْعِ رَسُوبٌ إذَا . مَا ثَاخَ فِي مُحْتَفَلٍ يَخْتَلِي وبنحو الذي قلنا في ذلك قال أهل التأويل.\nذكر من قال ذلك:\n⁕ حدثنا ابن حميد، قال: ثنا مهران، قال: ثنا سفيان، عن خصيف، عن عكرِمة، عن ابن عباس: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ قال: السحاب فيه المطر.\n⁕ حدثنا عليّ بن سهل، قال: ثنا مؤمل، قال: ثنا سفيان، عن خَصِيف، عن عكرِمة، عن ابن عباس في قوله: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ قال: ذات السحاب فيه المطر.\n⁕ حدثني محمد بن سعد، قال: ثني أبي، قال: ثني عمي، قال: ثني أبي، عن أبيه، عن ابن عباس: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ يعني بالرجع: القطر والرزق كلّ عام.\n⁕ حدثني يعقوب، قال: ثنا ابن عُلَية، عن أبي رجاء، عن الحسن، في قوله: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ قال: ترجع بأرزاق الناس كلّ عام؛ قال أبو رجاء: سُئل عنها عكرِمة، فقال: رجعت بالمطر.\n⁕ حدثني محمد بن عمرو، قال: ثنا أبو عاصم، قال: ثنا عيسى؛ وحدثني الحارث، قال: ثنا الحسن، قال: ثنا ورقاء، جميعا عن ابن أبي نجيح، عن مجاهد، قوله: ﴿ذَاتِ الرَّجْعِ﴾ قال: السحاب يمطر، ثم يَرجع بالمطر.\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة، قوله: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ قال: ترجع بأرزاق العباد كلّ عام، لولا ذلك هلَكوا، وهلَكت مواشيهم.\n⁕ حدثنا ابن عبد الأعلى، قال: ثنا ابن ثور، عن معمر، عن قتادة، قوله: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ قال: ترجع بالغيث كلّ عام.\n⁕ حُدثت عن الحسين: قال: سمعت أبا معاذ يقول: ثنا عبيد، قال: سمعت الضحاك يقول في قوله: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ يعني: المطر.\nوقال آخرون: يعني بذلك: أن شمسها وقمرها يغيب ويطلُع.\nذكر من قال ذلك:\n⁕ حدثني يونس، قال: أخبرنا ابن وهب، قال: قال ابن زيد، في قوله: ﴿وَالسَّمَاءِ ذَاتِ الرَّجْعِ﴾ قال: شمسها وقمرها ونجومها يأتين من هاهنا.\n* * وقوله: ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾\nيقول تعالى ذكره: والأرض ذات الصدع بالنبات.\nوبنحو الذي قلنا في ذلك قال أهل التأويل.\nذكر من قال ذلك:\n⁕ حدثنا ابن حميد، قال: ثنا مهران، عن سفيان، عن خَصِيف، عن عكرِمة، عن ابن عباس ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ قال: ذات النبات.\n⁕ حدثني محمد بن سعد، قال: ثني أبي، قال: ثني عمي، قال: ثني أبي، عن أبيه، عن ابن عباس: ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ يقول: صدعها إخراج النبات في كلّ عام.\n⁕ حدثني يعقوب، قال: ثنا ابن عُلَيَة، عن أبي رجاء، عن الحسن ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ قال: هذه تصدع عما تحتها؛ قال أبو رجاء: وسُئل عنها عكرِمة، فقال: هذه تصدع عن الرزق.\n⁕ حدثني محمد بن عمرو، قال: ثنا أبو عاصم، قال: ثنا عيسى؛ وحدثني الحارث، قال: ثنا الحسن، عن ابن أبي نجيح، قال مجاهد: ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ مثل المأزم مأزم منى.\n⁕ حدثني الحارث، قال: ثنا الحسن، قال: ثنا ورقاء، جميعا عن ابن أبي نجيح، عن مجاهد: ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ قال: الصدع: مثل المأزم، غير الأودية وغير الجُرُف.\n⁕ حدثنا بشر، قال: ثنا يزيد، قال: ثنا سعيد، عن قتادة، قوله: ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ تصدع عن الثمار وعن النبات كما رأيتم.\n⁕ حدثنا ابن عبد الأعلى، قال: ثنا ابن ثور، عن معمر، عن قتادة ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ قال: تصدع عن النبات.\n⁕ حدثني يونس، قال: أخبرنا ابن وهب، قال: قال ابن زيد، في قوله: ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ وقرأ: ﴿ثُمَّ شَقَقْنَا الأرْضَ شَقًّا فَأَنْبَتْنَا فِيهَا حَبًّا وَعِنَبًا وَقَضْبًا﴾ إلى آخر الآية، قال: صدعها للحرث.\n⁕ حُدثت عن الحسين، قال: سمعت أبا معاذ يقول: أخبرنا عبيد، قال: سمعت الضحاك يقول في قوله: ﴿وَالأرْضِ ذَاتِ الصَّدْعِ﴾ :النبات.\n* * وقوله: ﴿إِنَّهُ لَقَوْلٌ فَصْلٌ﴾\nيقول تعالى ذكره: إن هذا القول وهذا الخبر لقول فصل: يقول: لقول يفصل بين الحقّ والباطل ببيانه.\nوبنحو الذي قلنا في ذلك قال أهل التأويل على اختلاف منهم في العبارة عنه، فقال بعضهم: لقول حقّ. وقال بعضهم: لقول حكم.\nذكر من قال ذلك:\n⁕ حدثني عليّ، قال: ثنا أبو صالح، قال: ثني معاوية، عن عليّ،
